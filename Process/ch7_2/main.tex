\section{分辨率限制（Resolution Limits）}

\subsection{焦深（Depth of Focus, DOF）}

\subsubsection{焦深的基本概念}

\textbf{定义：}离开焦平面一定范围所成的像仍能够满足一定分辨率要求，这个范围就是焦深（DOF）。

\textbf{与焦距的区别：}
\begin{itemize}
    \item \textbf{焦距/焦点/焦平面：}成像最清晰的位置
    \item \textbf{焦深：}允许偏离焦平面的深度范围
\end{itemize}

\textbf{焦深公式：}
\begin{equation}
DOF = k_2 \frac{\lambda}{(NA)^2}
\end{equation}

其中：
\begin{itemize}
    \item $k_2$ = 工艺相关常数
    \item $\lambda$ = 曝光波长
    \item NA = 数值孔径
\end{itemize}

\textbf{另一种表述形式：}
\begin{equation}
DOF \approx \frac{\lambda}{2(NA)^2}
\end{equation}

\textbf{物理意义：}
\begin{itemize}
    \item 晶圆表面不完全平坦
    \item 需要足够的焦深容忍表面起伏
    \item 焦深越大，工艺容忍度越高
\end{itemize}

\subsubsection{光程差（OPD）与焦深}

\textbf{光程差定义：}光通过不同路径产生的相位差

\textbf{临界光程差：}
\begin{itemize}
    \item 当OPD达到一定值时，零级衍射光与一级衍射光相位相差90°
    \item 此时两者不再干涉，无法形成图形
    \item 实际可容忍的OPD必须小于这个临界值
\end{itemize}

\textbf{与DOF的关系：}OPD限制了可接受的焦深范围

\subsection{分辨率与焦深的权衡（Resolution vs. DOF）}

\subsubsection{基本矛盾}

\textbf{设计目标：}
\begin{itemize}
    \item 需要小的$l_{min}$（高分辨率）
    \item 需要大的DOF（高容忍度）
\end{itemize}

\textbf{物理限制：}
\begin{equation}
R = k_1\frac{\lambda}{NA}, \quad DOF = k_2\frac{\lambda}{(NA)^2}
\end{equation}

\textbf{矛盾分析：}
\begin{itemize}
    \item 提高分辨率需要增大NA $\rightarrow$ R减小
    \item 增大NA会导致DOF减小
    \item 两者存在本质上的权衡关系
\end{itemize}

\subsubsection{DOF的重要性}

\textbf{实际工艺考虑：}
\begin{itemize}
    \item \textbf{工艺拓扑：}台阶、不平整表面
    \item \textbf{晶圆平整度：}晶圆本身的翘曲
    \item \textbf{设备控制：}自动对焦误差
\end{itemize}

\textbf{DOF要求：}
\begin{itemize}
    \item 如果DOF < $\Delta$（表面起伏），则无法获得清晰图像
    \item 必须满足：DOF $\geq$ $\Delta$
    \item 或通过CMP等工艺改善表面平整度
\end{itemize}

\section{分辨率增强技术（Resolution Enhancement Techniques, RET）}

\subsection{RET技术概述}

\subsubsection{技术背景}

\textbf{现状：}近年来，临界尺寸显著小于光源波长

\textbf{示例：}
\begin{itemize}
    \item 光源波长：193 nm
    \item 特征尺寸：< 50 nm
    \item 尺寸/波长比：< 1/4
\end{itemize}

\textbf{解决方案：}使用分辨率增强技术（tricks）突破衍射极限

\subsubsection{RET技术分类}

\textbf{主要技术：}
\begin{enumerate}
    \item \textbf{离轴照明（Off-Axis Illumination, OAI）}
    \item \textbf{相移掩模（Phase Shift Mask, PSM）}
    \item \textbf{光学邻近修正（Optical Proximity Correction, OPC）}
    \item \textbf{亚分辨率辅助特征（Sub-Resolution Assist Features, SRAF）}
    \item \textbf{双重图案化（Double Patterning）}
    \item \textbf{干涉光刻（Interferometric Lithography）}
\end{enumerate}

\textbf{技术路线：}
\begin{itemize}
    \item 减小波长$\lambda$
    \item 增大数值孔径NA
    \item 减小$k_1$因子
\end{itemize}

\subsection{减小波长$\lambda$}

\subsubsection{波长与分辨率的关系}

\textbf{基本原理：}
\begin{equation}
l_{min} = k_1 \frac{\lambda}{NA}
\end{equation}

\textbf{物理意义：}
\begin{itemize}
    \item 显微镜：波长越短，看到更小特征
    \item 光刻：波长越短，写出更小图形
\end{itemize}

\textbf{光子能量与波长：}
\begin{equation}
E = h\nu = \frac{hc}{\lambda} = \frac{1239.842 \text{ eV·nm}}{\lambda}
\end{equation}

波长越短，光子能量越高

\subsubsection{电磁波谱}

\textbf{光刻使用的波长：}
\begin{itemize}
    \item \textbf{G线：}436 nm
    \item \textbf{H线：}405 nm
    \item \textbf{I线：}365 nm
    \item \textbf{KrF：}248 nm
    \item \textbf{ArF：}193 nm
    \item \textbf{F$_2$：}157 nm
    \item \textbf{EUV：}13.5 nm
\end{itemize}

\textbf{波长缩短带来的改善：}
\begin{itemize}
    \item 每次波长缩短可带来相应比例的分辨率提升
\end{itemize}

\subsubsection{短波长技术挑战}

\textbf{F$_2$光刻（$\lambda$ = 157 nm）：}

\textbf{掩模材料：}改性熔融硅（Modified fused silica）

\textbf{主要挑战：}
\begin{enumerate}
    \item \textbf{无防护膜（No pellicle）：}掩模保护困难
    \item \textbf{透镜材料：}CaF$_2$（氟化钙）
    \begin{itemize}
        \item 光学质量差
        \item 制造困难
    \end{itemize}
    \item \textbf{光刻胶材料：}需要专门开发
    \item \textbf{成本：}极高
\end{enumerate}

\textbf{发展现状：}
\begin{itemize}
    \item 2010年：193 nm浸没式光刻实现32 nm制程（Intel Core i7）
    \item 结合双重成像技术延伸到22 nm（2013）、14 nm（2017）
    \item F$_2$光刻未成为主流
\end{itemize}

\subsubsection{波长缩短的历史进程}

\textbf{分辨率提升百分比：}随着波长从436 nm缩短到13.5 nm，分辨率提升超过30倍

\subsection{增大数值孔径NA}

\subsubsection{数值孔径的定义}

\textbf{公式：}
\begin{equation}
NA = n \sin(\alpha)
\end{equation}

其中：
\begin{itemize}
    \item n = 介质折射率
    \item $\alpha$ = 半孔径角
\end{itemize}

\subsubsection{增大NA的方法}

\textbf{方法一：增大孔径角$\alpha$}

\textbf{进展：}
\begin{itemize}
    \item 透镜设计/制造允许NA接近1
    \item 已实现NA = 0.93
\end{itemize}

\textbf{高NA问题：}
\begin{itemize}
    \item \textbf{小DOF：}需要超薄光刻胶工艺
    \item \textbf{偏振和双折射：}光学系统内的偏振效应
\end{itemize}

\textbf{方法二：增大折射率n（浸没式光刻）}

\subsection{浸没式光刻（Immersion Lithography）}

\subsubsection{基本原理}

\textbf{概念：}在透镜与晶圆之间填充高折射率液体

\textbf{折射率数据：}
\begin{itemize}
    \item $n_{air}$ = 1
    \item $n_{water}$ = 1.44（在193 nm波长）
\end{itemize}

\textbf{效果：}
\begin{equation}
NA = n \sin(\alpha)
\end{equation}

使用水作为介质，NA可以超过1

\subsubsection{分辨率计算示例}

\textbf{给定条件：}
\begin{itemize}
    \item $k_1$ = 0.25
    \item $\lambda$ = 193 nm
    \item NA = 0.93
\end{itemize}

\textbf{空气中：}
\begin{equation}
l_{min} = \frac{0.25 \times 193}{0.93} = 51.9 \text{ nm}
\end{equation}

\textbf{水中（$n_{water}$ = 1.44）：}
\begin{equation}
l_{min} = \frac{0.25 \times 193}{0.93 \times 1.44} = 36.0 \text{ nm}
\end{equation}

可实现约45 nm特征

\subsubsection{浸没式光刻的优势}

\textbf{主要优势：}
\begin{enumerate}
    \item \textbf{突破NA > 1限制：}可达到$n_{fluid}$允许的范围
    \item \textbf{DOF改善：}
    \begin{equation}
    DOF = k_2\frac{\lambda}{(NA)^2}
    \end{equation}
    DOF改善幅度超过n倍
    \item \textbf{H$_2$O是193 nm的良好选择：}$n_{water} \approx 1.44$
\end{enumerate}

\subsubsection{浸没式光刻的挑战}

\textbf{技术挑战：}
\begin{enumerate}
    \item \textbf{光学成本和复杂性：}系统设计复杂
    \item \textbf{缺陷：}液体相关缺陷
    \item \textbf{污染：}液体污染控制
    \item \textbf{流体成本：}高纯度液体价格高
\end{enumerate}

\subsubsection{DOF趋势}

\textbf{历史趋势：}
\begin{itemize}
    \item NA接近1时，DOF $\approx$ $\lambda$/2
    \item DOF缩放主导光刻技术演进
\end{itemize}

\textbf{改善因素：}
\begin{itemize}
    \item 光学技术改进超预期
    \item 透镜设计/制造进步
    \item 先进工艺减少焦点预算：
    \begin{itemize}
        \item 工艺拓扑改善（CMP）
        \item 晶圆平整度提升
        \item 工具控制、自动对焦误差减小
    \end{itemize}
\end{itemize}

\subsubsection{扩展浸没式光刻}

\textbf{更高折射率液体：}

\textbf{研究方向：}开发折射率更高的液体

\textbf{示例材料：}
\begin{itemize}
    \item Sumitomo开发的高折射率液体
    \item IMEC研究的材料
\end{itemize}

\textbf{目标：}进一步提高NA和分辨率

\subsection{减小$k_1$因子}

\subsubsection{$k_1$因子的意义}

\textbf{分辨率公式：}
\begin{equation}
l_{min} = k_1 \frac{\lambda}{NA}
\end{equation}

\textbf{$k_1$的影响因素：}
\begin{itemize}
    \item \textbf{光学系统：}
    \begin{itemize}
        \item 透镜质量（像差等）
        \item 照明条件
        \item 孔径平面的相干度和强度分布
    \end{itemize}
    \item \textbf{掩模：}掩模设计和质量
    \item \textbf{光刻胶和工艺窗口：}工艺容忍度
\end{itemize}

\textbf{简化定义：}
\begin{itemize}
    \item $l_{min}$通常取周期光栅间距P的一半
    \item $k_1$是上述因素的综合函数
\end{itemize}

\subsubsection{减小$k_1$的关键}

\textbf{三大要素：}
\begin{enumerate}
    \item \textbf{低像差光学系统：}高质量透镜
    \item \textbf{优秀的工艺控制：}稳定的工艺参数
    \item \textbf{精确的掩模图形：}高质量掩模制作
\end{enumerate}

\subsubsection{光刻技术趋势}

\textbf{$k_1$、$\lambda$、NA的演进：}

\textbf{历史数据（所有尺寸单位：nm）：}
\begin{itemize}
    \item 持续减小$k_1$
    \item 持续减小$\lambda$
    \item 持续增大NA
    \item 三者协同作用实现$l_{min}$的持续缩小
\end{itemize}

\subsection{相移掩模（Phase Shift Mask, PSM）}

\subsubsection{基本原理}

\textbf{光学基础：}

\textbf{电场（$\varepsilon$）：}
\begin{itemize}
    \item 光波的电场分量
\end{itemize}

\textbf{强度（I）：}
\begin{equation}
I \propto |\varepsilon|^2
\end{equation}

\textbf{能量（E）：}
\begin{equation}
E = \int I \, dt
\end{equation}

\subsubsection{相移原理}

\textbf{核心思想：}通过一透明区的光波与通过相邻透明区但具有180°相位差的光波之间的干涉作用

\textbf{效果：}
\begin{itemize}
    \item 使相邻透明区边界处的光强减弱
    \item 图形边沿对比度得到提高
\end{itemize}

\subsubsection{PSM类型}

\textbf{1. 衰减型PSM（Attenuated PSM）}

\textbf{2. 交替型PSM（Alternating PSM）}

\subsubsection{PSM应用实例}

\textbf{DUV光刻应用：}
\begin{itemize}
    \item 193 nm深紫外光刻结合PSM
    \item 可实现60 nm线宽
    \item Lucent等公司的应用案例
\end{itemize}

\subsection{光学邻近修正（Optical Proximity Correction, OPC）}

\subsubsection{OPC的必要性}

\textbf{实际光刻问题：}

\textbf{CAD设计与实际工艺差异：}
\begin{itemize}
    \item \textbf{拐角圆化（Corner rounding）}
    \item \textbf{线条缩短（Line shortening）}
    \item \textbf{套刻误差（Overlay errors）}
    \item \textbf{临界尺寸误差（CD errors）}
    \item \textbf{弯曲和扭曲（Bends and wiggles）}
\end{itemize}

\subsubsection{OPC基本原理}

\textbf{定义：}在光刻版上进行图形修正，补偿衍射带来的光刻图形变形

\textbf{方法：}有意地扭曲掩模图形，以补偿系统性和稳定的图形误差

\subsubsection{OPC技术演进}

\textbf{历史进程：}
\begin{enumerate}
    \item \textbf{DRAM和SRAM单元的"调整"：}早期手工调整
    \item \textbf{锚点形状（Anchor shapes）：}补偿栅极缩短
    \item \textbf{基于规则的OPC（Rules-based OPC）：}
    \begin{itemize}
        \item 根据修正表选择性偏置线宽
        \item 使用设计规则检查器实现
    \end{itemize}
    \item \textbf{基于模型的OPC（Model-based OPC）：}
    \begin{itemize}
        \item 迭代修改形状
        \item 直到模型符合容差要求
    \end{itemize}
\end{enumerate}

\subsubsection{OPC效果对比}

\textbf{无OPC：}图形严重变形

\textbf{有OPC：}图形接近设计目标

\textbf{应用：}全工艺补偿（Total Process Compensation）

\subsection{亚分辨率辅助特征（Sub-Resolution Assist Features, SRAF）}

\subsubsection{SRAF原理}

\textbf{设计思想：}
\begin{itemize}
    \item 主特征以更大DOF印刷
    \item 辅助特征不应印刷出来
    \item 孤立线和密集线变得更加均匀
\end{itemize}

\textbf{效果：}
\begin{itemize}
    \item 改善光刻均匀性
    \item 增大工艺窗口
    \item 减少邻近效应
\end{itemize}

\subsubsection{SRAF应用}

\textbf{应用场景：}
\begin{itemize}
    \item 孤立线条周围添加不可见辅助线
    \item 使孤立线与密集线的曝光条件趋于一致
\end{itemize}

\subsection{双重图案化（Double Patterning）}

\subsubsection{双重图案化原理}

\textbf{基本方法：}
\begin{itemize}
    \item 双次曝光 + 双次刻蚀工艺
    \item 两个光栅（1:3间距比）交错排列
    \item 注意：简单的双次曝光不可行
\end{itemize}

\textbf{有效$k_1$降低：}
\begin{itemize}
    \item 实际印刷半间距$k_1$ = 0.15
    \item 两个$k_1$ = 0.3的光栅组合
\end{itemize}

\textbf{挑战：}
\begin{itemize}
    \item 套刻误差
    \item 工艺变形
\end{itemize}

\subsubsection{双重图案化实例}

\textbf{技术示例：}
\begin{itemize}
    \item 180 nm间距光栅：由两次360 nm光栅曝光实现
    \item Brueck et al., SPIE 2002
\end{itemize}

\subsubsection{双重图案化的变体}

\textbf{1. 光刻胶修整（Resist Trim）}

\textbf{IMEC技术：}
\begin{itemize}
    \item 通过修整光刻胶减小线宽
    \item 进一步提高分辨率
\end{itemize}

\textbf{2. 双重图案化 + OPC}

\textbf{综合技术：}
\begin{itemize}
    \item IMEC的综合方案
    \item 结合双重图案和光学邻近修正
\end{itemize}

\textbf{3. 灰化工艺、侧壁工艺}

\textbf{目的：}降低$k_1$

\textbf{方法：}
\begin{itemize}
    \item 通过灰化减小特征尺寸
    \item 侧壁间隔物技术
\end{itemize}

\subsection{干涉光刻（Interferometric Lithography）}

\subsubsection{干涉光刻原理}

\textbf{光栅间距公式：}
\begin{equation}
P_L = \frac{\lambda}{2\sin\alpha}
\end{equation}

其中：
\begin{itemize}
    \item $P_L$ = 光栅间距
    \item $\lambda$ = 波长
    \item $\alpha$ = 入射角
\end{itemize}

\textbf{特点：}
\begin{itemize}
    \item \textbf{无掩模（Maskless）}
    \item \textbf{仅依赖$\lambda$和$\alpha$}
    \item \textbf{仅限周期性结构：}无法实现任意特征
\end{itemize}

\textbf{归属：}PSM、双重掩膜、OPC、驻波效应校正属于波前工程突破

\subsubsection{干涉光刻应用}

\textbf{应用示例：}
\begin{itemize}
    \item 制作周期性光栅
    \item 纳米尺度周期结构
\end{itemize}

\subsubsection{浸没式干涉光刻}

\textbf{NEMO系统（IBM Almaden）：}
\begin{itemize}
    \item 浸没式 + 干涉光刻
    \item 实现30 nm半间距
\end{itemize}

\textbf{优势：}
\begin{itemize}
    \item 结合浸没式的高NA
    \item 结合干涉的高分辨率
    \item 适合研究和特殊应用
\end{itemize}

\section{极紫外光刻（EUV Lithography）}

\subsection{EUV光刻概述}

\subsubsection{EUV基本参数}

\textbf{波长：}$\lambda$ = 13.5 nm

\textbf{技术特点：}
\begin{itemize}
    \item 极短波长
    \item 突破传统光学光刻极限
    \item 下一代主流光刻技术
\end{itemize}

\textbf{应用节点：}
\begin{itemize}
    \item 7 nm及以下制程
    \item 5 nm、3 nm等先进节点
\end{itemize}

\subsection{EUV光源}

\subsubsection{EUV光源要求}

\textbf{功率要求：}
\begin{itemize}
    \item 中间焦点功率 > 180 W
    \item 可接受的实用性要求
\end{itemize}

\textbf{性能指标：}
\begin{itemize}
    \item 通过提高转换效率

\subsection{重要公式和数据}

\begin{enumerate}
    \item \textbf{焦深：}DOF = $k_2\lambda/(NA)^2$ 或 DOF $\approx$ $\lambda/[2(NA)^2]$
    \item \textbf{分辨率：}$l_{min} = k_1\lambda/NA$
    \item \textbf{数值孔径：}NA = $n\sin(\alpha)$
    \item \textbf{光子能量：}$E = hc/\lambda = 1239.842$ eV·nm$/\lambda$
    \item \textbf{浸没式分辨率改善：}空气51.9 nm $\rightarrow$ 水36.0 nm（$k_1$=0.25, $\lambda$=193nm, NA=0.93）
    \item \textbf{干涉光栅间距：}$P_L = \lambda/(2\sin\alpha)$
    \item \textbf{波长演进：}G线436nm、I线365nm、KrF 248nm、ArF 193nm、F$_2$ 157nm、EUV 13.5nm
    \item \textbf{水的折射率：}$n_{water}$ = 1.44（193 nm）
    \item \textbf{实现的NA：}0.93（干式）；>1（浸没式）
    \item \textbf{双重图案$k_1$：}实际0.15（两个0.3光栅组合）
    \item \textbf{EUV多层膜精度：}$\Delta t < \lambda$/500
    \item \textbf{EUV光学精度：}面形误差 < 0.10 nm RMS；缺陷 $\sim$ $\lambda$/50
\end{enumerate}

\subsection{常见考点和易错点}

\begin{enumerate}
    \item \textbf{DOF与分辨率的矛盾}
    \begin{itemize}
        \item 增大NA：提高分辨率但减小DOF
        \item 需要权衡或通过其他技术补偿（如CMP）
    \end{itemize}
    
    \item \textbf{浸没式光刻的优势}
    \begin{itemize}
        \item 不仅是分辨率提高n倍
        \item DOF改善幅度超过n倍（因为分母是$(NA)^2$）
    \end{itemize}
    
    \item \textbf{$k_1$因子的意义}
    \begin{itemize}
        \item 不是固定常数
        \item 取决于光学、掩模、工艺的综合水平
        \item 技术进步体现在$k_1$的持续降低
    \end{itemize}
    
    \item \textbf{PSM的工作原理}
    \begin{itemize}
        \item 利用相位差（180°）
        \item 干涉使边界光强减弱
        \item 提高对比度而非直接提高分辨率
    \end{itemize}
    
    \item \textbf{OPC的本质}
    \begin{itemize}
        \item 预畸变补偿
        \item 掩模图形≠期望图形
        \item 最终晶圆图形≈期望图形
    \end{itemize}
    
    \item \textbf{双重图案化的限制}
    \begin{itemize}
        \item 仅适用于特定图形（可分解的）
        \item 套刻误差影响严重
        \item 不能简单双次曝光
    \end{itemize}
    
    \item \textbf{干涉光刻的局限}
    \begin{itemize}
        \item 无掩模是优点也是缺点
        \item 仅能形成周期结构
        \item 不能形成任意图形
    \end{itemize}
    
    \item \textbf{EUV的特殊性}
    \begin{itemize}
        \item 全反射光学（不是透射）
        \item 反射式掩模（不是透射式）
        \item 真空环境（空气吸收EUV）
        \item 无防护膜（材料限制）
    \end{itemize}
\end{enumerate}

\textbf{典型计算题示例：}

\textbf{题目1（作业题）：}对于浸入式光刻，计算浸入不同液体介质时光刻能达到的最小线宽。
\begin{itemize}
    \item 条件1：$n$ = 1.63，$\lambda$ = 193 nm，$k_1$ = 0.25，NA = 0.93（空气中）
    \item 条件2：$n$ = 1，$\lambda$ = 157 nm，$k_1$ = 0.25
\end{itemize}

\textbf{解答思路：}
\begin{itemize}
    \item 浸没后：$NA_{imm} = n \times NA_{air} = 1.63 \times 0.93 = 1.516$
    \item $l_{min} = k_1\lambda/NA_{imm} = 0.25 \times 193 / 1.516 = 31.8$ nm
    \item 比较F$_2$光刻（157 nm干式）的分辨率
    \item 结论：浸没式193 nm优于干式157 nm
\end{itemize}

\textbf{题目2（作业题）：}如果DOF小于$\Delta$（表面起伏），意味着什么？若希望得到清晰图像，$\Delta$必须满足什么条件？

\textbf{解答：}
\begin{itemize}
    \item 意味着：无法在整个表面范围内获得清晰成像
    \item 必须满足：$\Delta \leq DOF$
    \item 或者通过CMP等工艺减小$\Delta$
\end{itemize}

\subsection{记忆技巧}

\begin{itemize}
    \item \textbf{DOF公式：}DOF $\propto$ $\lambda/(NA)^2$，分母是NA的平方
    \item \textbf{RET六大技术：}离OPC（离轴、OPC）、相双干（相移、双重、干涉）、SRAF
    \item \textbf{波长记忆：}436-365-248-193-157-13.5，每次约减半或1/3
    \item \textbf{浸没优势：}分辨率提高n倍，DOF提高超过n倍（因为分母是平方）
    \item \textbf{PSM核心：}180°相位差$\rightarrow$破坏性干涉$\rightarrow$边界暗
    \item \textbf{OPC演进：}手工$\rightarrow$规则$\rightarrow$模型，越来越智能
    \item \textbf{SRAF原则：}主印辅不印，孤立变密集
    \item \textbf{双重图案：}双曝双刻，间距减半，套刻关键
    \item \textbf{EUV特征：}13.5 nm，全反射，反射掩模，无防护膜
    \item \textbf{EUV挑战：}光源功率、掩模缺陷、光刻胶、污染控制、成本
\end{itemize}

\subsection{综合复习要点}

\textbf{三大核心问题：}
\begin{enumerate}
    \item \textbf{如何提高分辨率？}
    \begin{itemize}
        \item 减小$\lambda$（波长演进）
        \item 增大NA（浸没式）
        \item 减小$k_1$（各种RET技术）
    \end{itemize}
    
    \item \textbf{如何平衡分辨率和DOF？}
    \begin{itemize}
        \item 优化工艺（CMP平坦化）
        \item 先进对焦系统
        \item 薄光刻胶工艺
        \item 浸没式（DOF改善超过n）
    \end{itemize}
    
    \item \textbf{为什么需要EUV？}
    \begin{itemize}
        \item 193 nm即使加RET也接近极限
        \item 多重图案成本和复杂度过高
        \item EUV提供本质上的分辨率提升
        \item 简化工艺流程（单次曝光 vs 多次图案化）
    \end{itemize}
\end{enumerate}

\textbf{学习建议：}
\begin{itemize}
    \item 先掌握DOF和分辨率的公式及其物理意义

    \item 重点理解浸没式光刻和EUV（产业主流）
    \item 结合第一部分内容，形成完整知识体系
    \item 关注技术演进逻辑和产业应用实例
    \item 练习计算题，掌握公式应用
\end{itemize}
