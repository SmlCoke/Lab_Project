\begin{algorithm}[H]%
	\caption{RAND-MIN-CUT($G=(V,E)$)}
	\begin{algorithmic}[1]
		\State 为图$G$的所有边都随机赋予一个正权值
		\State 找出$G$的最小生成树$T$
		\State 删除$T$中权值最大的一条边得到两棵子树$T_1,~T_2$
		\State 令$T_1$的顶点集为$V_1$，则$T_2$的顶点集为$V_2=V-V_1$
		\State $C=\{ uv|uv\in E, u\in V_1, v\in V_2 \}$
		\State \Return{$C$}
	\end{algorithmic}
\end{algorithm}
